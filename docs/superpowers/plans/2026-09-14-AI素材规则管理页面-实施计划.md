    @Override
    public String querySnapshot(String sourceLane, String materialType, String suggestType) {
        LambdaQueryChainWrapper<AiMaterialSuggestionRule> query = lambdaQuery();
        if (StringUtils.isNotBlank(sourceLane)) {
            query.in(AiMaterialSuggestionRule::getSourceLane,
                    java.util.Arrays.asList(sourceLane, AiMaterialSuggestionConstants.SOURCE_LANE_ALL));
        }
        List<AiMaterialSuggestionRule> rules = query
                .eq(AiMaterialSuggestionRule::getMaterialType, materialType)
                .eq(AiMaterialSuggestionRule::getSuggestType, suggestType)
                .eq(AiMaterialSuggestionRule::getEnabled, 1)
                .orderByDesc(AiMaterialSuggestionRule::getPriority)
                .list();
        if (rules.isEmpty()) {
            return "[]";
        }
        JSONArray array = new JSONArray();
        for (AiMaterialSuggestionRule rule : rules) {

            obj.put("ruleCode", rule.getRuleCode());
            obj.put("ruleName", rule.getRuleName());
            obj.put("ruleSource", rule.getRuleSource());
            obj.put("ruleType", rule.getRuleType());
            obj.put("ruleContent", rule.getRuleContent());
            obj.put("priority", rule.getPriority());

    private int changeEnabled(AiSuggestionRuleBatchReqDTO req, String operator, String source, int enabled, String action) {
        boolean requireFilter = SOURCE_PAGE.equals(source);
        List<AiMaterialSuggestionRule> targets = queryTargetRules(req, requireFilter);
        long now = System.currentTimeMillis();
        int count = 0;
        for (AiMaterialSuggestionRule rule : targets) {
            if (rule.getEnabled() != null && rule.getEnabled() == enabled) {
                continue;
            }
            AiSuggestionRuleFieldChangeDTO change = new AiSuggestionRuleFieldChangeDTO()
                    .setField("enabled")
                    .setFrom(rule.getEnabled())
                    .setTo(enabled);
            rule.setEnabled(enabled);
            rule.setUpdateTime(now);
            updateById(rule);
            writeLog(rule.getRuleCode(), action, source, operator, Collections.singletonList(change));
            count++;
        }
        return count;
    }

    private List<AiMaterialSuggestionRule> queryTargetRules(AiSuggestionRuleBatchReqDTO req, boolean requireFilter) {
        if (requireFilter && allFiltersEmpty(req)) {
            throw new ServiceException("At least one id/filter is required");
        }
        if (req.getIds() != null && !req.getIds().isEmpty()) {
            List<AiMaterialSuggestionRule> result = new ArrayList<>();
            for (List<Long> chunk : chunk(req.getIds(), IN_BATCH_SIZE)) {
                result.addAll(lambdaQuery().in(AiMaterialSuggestionRule::getId, chunk).list());
            }
            return result;
        }
        return lambdaQuery()
                .in(req.getRuleCodes() != null && !req.getRuleCodes().isEmpty(),
                        AiMaterialSuggestionRule::getRuleCode, req.getRuleCodes())
                .in(req.getSuggestTypes() != null && !req.getSuggestTypes().isEmpty(),
                        AiMaterialSuggestionRule::getSuggestType, req.getSuggestTypes())
                .in(req.getSourceLanes() != null && !req.getSourceLanes().isEmpty(),
                        AiMaterialSuggestionRule::getSourceLane, req.getSourceLanes())
                .list();
    }

    private boolean allFiltersEmpty(AiSuggestionRuleBatchReqDTO req) {
        return (req.getIds() == null || req.getIds().isEmpty())
                && (req.getRuleCodes() == null || req.getRuleCodes().isEmpty())
                && (req.getSuggestTypes() == null || req.getSuggestTypes().isEmpty())
                && (req.getSourceLanes() == null || req.getSourceLanes().isEmpty());
    }

    private <T> List<List<T>> chunk(List<T> list, int size) {
        List<List<T>> chunks = new ArrayList<>();
        for (int i = 0; i < list.size(); i += size) {
            chunks.add(list.subList(i, Math.min(i + size, list.size())));
        }
        return chunks;
    }

            JSONObject obj = array.getJSONObject(i);
            String ruleCode = obj.getString("ruleCode");
            if (StringUtils.isBlank(ruleCode)) {
                continue;
            }
            AiMaterialSuggestionRule existing = lambdaQuery()
                    .eq(AiMaterialSuggestionRule::getRuleCode, ruleCode)
                    .one();
            if (existing != null) {
                AiMaterialSuggestionRule before = copyRule(existing);

                existing.setSourceLane(obj.getString("sourceLane"));
                existing.setMaterialType(obj.getString("materialType"));
                existing.setSuggestType(obj.getString("suggestType"));
                existing.setRuleSource(obj.getString("ruleSource"));
                existing.setRuleType(obj.getString("ruleType"));
                existing.setRuleContent(obj.getString("ruleContent"));
                existing.setPriority(obj.getInteger("priority"));
                existing.setEnabled(obj.getIntValue("enabled"));
                existing.setOperator(obj.getString("operator"));
                existing.setUpdateTime(now);
                updateById(existing);
                List<AiSuggestionRuleFieldChangeDTO> changes = buildFieldChanges(before, existing);
                if (!changes.isEmpty()) {
                    writeLog(ruleCode, ACTION_UPDATE, source, operator, changes);
                }
            } else {
                AiMaterialSuggestionRule rule = new AiMaterialSuggestionRule()
                        .setRuleCode(ruleCode)
                        .setRuleName(obj.getString("ruleName"))
                        .setSourceLane(obj.getString("sourceLane"))
                        .setMaterialType(obj.getString("materialType"))
                        .setSuggestType(obj.getString("suggestType"))
                        .setRuleSource(obj.getString("ruleSource"))
                        .setRuleType(obj.getString("ruleType"))
                        .setRuleContent(obj.getString("ruleContent"))
                        .setPriority(obj.getInteger("priority"))
                        .setEnabled(obj.getIntValue("enabled"))
                        .setOperator(obj.getString("operator"))
                        .setCreateTime(now)
                        .setUpdateTime(now);
                save(rule);
                writeLog(ruleCode, ACTION_CREATE, source, operator, buildCreateChanges(rule));
            }
            count++;
        }

    private List<AiSuggestionRuleFieldChangeDTO> parseChanges(String changeDetail) {
        if (StringUtils.isBlank(changeDetail)) {
            return Collections.emptyList();
        }
        JSONArray array = JSON.parseArray(changeDetail);
        if (array == null) {
            return Collections.emptyList();
        }
        List<AiSuggestionRuleFieldChangeDTO> result = new ArrayList<>();
        for (int i = 0; i < array.size(); i++) {
            JSONObject obj = array.getJSONObject(i);
            result.add(new AiSuggestionRuleFieldChangeDTO()
                    .setField(obj.getString("field"))
                    .setFrom(obj.get("from"))
                    .setTo(obj.get("to")));
        }
        return result;
    }

        addChange(changes, "priority", before.getPriority(), after.getPriority());
        addChange(changes, "enabled", before.getEnabled(), after.getEnabled());
        addChange(changes, "operator", before.getOperator(), after.getOperator());

    private void addChange(List<AiSuggestionRuleFieldChangeDTO> changes, String field, Object from, Object to) {
        if (from == null && to == null) {
            return;
        }
        if (from != null && from.equals(to)) {
            return;
        }
        changes.add(new AiSuggestionRuleFieldChangeDTO().setField(field).setFrom(from).setTo(to));
    }

                          List<AiSuggestionRuleFieldChangeDTO> changes) {

                .setAction(action)
                .setSource(source)
                .setOperator(operator)
                .setChangeDetail(changes == null ? JSON.toJSONString(Collections.emptyList())
                        : JSON.toJSONString(changes));
        long now = System.currentTimeMillis();
        log.setCreateTime(now).setUpdateTime(now);
        ruleLogMapper.insert(log);
    }

    private void validateSave(AiSuggestionRuleSaveReqDTO req) {
        if (!AiMaterialSuggestionConstants.SOURCE_LANE_ALL.equals(req.getSourceLane())

            throw new ServiceException("invalid sourceLane: " + req.getSourceLane());
        }
        if (!"BODY".equals(req.getMaterialType())
                && !"HEADER".equals(req.getMaterialType())
                && !"FOOTER".equals(req.getMaterialType())) {
            throw new ServiceException("invalid materialType: " + req.getMaterialType());
        }
        if (SuggestTypeEnum.fromCode(req.getSuggestType()) == null) {
            throw new ServiceException("invalid suggestType: " + req.getSuggestType());
        }
        if (RuleSourceEnum.fromCode(req.getRuleSource()) == null) {
            throw new ServiceException("invalid ruleSource: " + req.getRuleSource());
        }
        if (RuleTypeEnum.fromCode(req.getRuleType()) == null) {
            throw new ServiceException("invalid ruleType: " + req.getRuleType());
        }
        if (StringUtils.isBlank(req.getRuleContent())) {
            throw new ServiceException("ruleContent cannot be blank");
        }
        try {
            Object parsed = JSON.parse(req.getRuleContent());
            if (!(parsed instanceof JSONObject)) {
                throw new ServiceException("ruleContent must be a JSON object");
            }
        } catch (ServiceException e) {
            throw e;
        } catch (Exception e) {
            throw new ServiceException("ruleContent is not valid JSON");
        }
    }

package com.opay.occ.whatsapp.data.ai.dto;

import com.alibaba.fastjson.JSON;
import com.alibaba.fastjson.JSONObject;
import com.opay.occ.whatsapp.data.entity.dto.request.AiSuggestionRuleBatchReqDTO;
import lombok.Data;
import org.apache.commons.lang3.StringUtils;

/**
 * Parser for AiMaterialSuggestionRuleImportJob param.
 * Param examples:
 *   "" or {"action":"IMPORT"}
 *   {"action":"ENABLE","ruleCodes":["R1"],"suggestTypes":[],"sourceLanes":[]}
 *   {"action":"DISABLE","ruleCodes":[],"suggestTypes":["GENERAL"],"sourceLanes":[]}
 */
@Data
public class AiMaterialSuggestionRuleJobCommand {

    public static final String ACTION_IMPORT = "IMPORT";
    public static final String ACTION_ENABLE = "ENABLE";
    public static final String ACTION_DISABLE = "DISABLE";

    private String action = ACTION_IMPORT;

    private AiSuggestionRuleBatchReqDTO batch = new AiSuggestionRuleBatchReqDTO();

    public static AiMaterialSuggestionRuleJobCommand parse(String param) {
        AiMaterialSuggestionRuleJobCommand cmd = new AiMaterialSuggestionRuleJobCommand();
        if (StringUtils.isBlank(param)) {
            return cmd;
        }
        JSONObject obj;
        try {
            obj = JSON.parseObject(param);
        } catch (Exception e) {
            return cmd;
        }
        if (obj == null) {
            return cmd;
        }
        String action = obj.getString("action");
        if (StringUtils.isNotBlank(action)) {
            cmd.setAction(action.toUpperCase());
        }
        AiSuggestionRuleBatchReqDTO batch = obj.toJavaObject(AiSuggestionRuleBatchReqDTO.class);
        if (batch != null) {
            cmd.setBatch(batch);
        }
        return cmd;
    }
}

package com.opay.occ.whatsapp.job.job.task;

import com.opay.occ.whatsapp.common.MDCUtil;
import com.opay.occ.whatsapp.common.config.BusinessConfig;
import com.opay.occ.whatsapp.common.constants.AiMaterialSuggestionConstants;
import com.opay.occ.whatsapp.data.ai.dto.AiMaterialSuggestionRuleJobCommand;
import com.opay.occ.whatsapp.data.entity.dto.request.AiSuggestionRuleBatchReqDTO;
import com.opay.occ.whatsapp.data.service.AiMaterialSuggestionRuleService;
import com.xxl.job.core.biz.model.ReturnT;
import com.xxl.job.core.handler.IJobHandler;
import com.xxl.job.core.handler.annotation.JobHandler;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import javax.annotation.Resource;

/**
 * XXL-Job (low-version compatible): Import/enable/disable AI material suggestion rules.
 * param supports:
 *   empty -> IMPORT from Apollo config
 *   {"action":"IMPORT"}
 *   {"action":"ENABLE","ruleCodes":[],"suggestTypes":[],"sourceLanes":[]}
 *   {"action":"DISABLE","ruleCodes":[],"suggestTypes":[],"sourceLanes":[]}
 */
@Slf4j
@Component
@JobHandler(value = "AiMaterialSuggestionRuleImportJob")
public class AiMaterialSuggestionRuleImportJob extends IJobHandler {

    @Resource
    private BusinessConfig businessConfig;

    @Resource
    private AiMaterialSuggestionRuleService aiMaterialSuggestionRuleService;

    @Override
    public ReturnT<String> execute(String param) {
        MDCUtil.start();
        log.info("AiMaterialSuggestionRuleImportJob started, param={}", param);
        try {
            AiMaterialSuggestionRuleJobCommand cmd = AiMaterialSuggestionRuleJobCommand.parse(param);
            if (AiMaterialSuggestionRuleJobCommand.ACTION_ENABLE.equals(cmd.getAction())) {
                AiSuggestionRuleBatchReqDTO batch = cmd.getBatch() == null ? new AiSuggestionRuleBatchReqDTO() : cmd.getBatch();
                int n = aiMaterialSuggestionRuleService.enableRules(batch, AiMaterialSuggestionConstants.SYSTEM_OPERATOR, "JOB");
                log.info("AiMaterialSuggestionRuleImportJob enable completed, count={}", n);
                return ReturnT.SUCCESS;
            }
            if (AiMaterialSuggestionRuleJobCommand.ACTION_DISABLE.equals(cmd.getAction())) {
                AiSuggestionRuleBatchReqDTO batch = cmd.getBatch() == null ? new AiSuggestionRuleBatchReqDTO() : cmd.getBatch();
                int n = aiMaterialSuggestionRuleService.disableRules(batch, AiMaterialSuggestionConstants.SYSTEM_OPERATOR, "JOB");
                log.info("AiMaterialSuggestionRuleImportJob disable completed, count={}", n);
                return ReturnT.SUCCESS;
            }
            String json = businessConfig.getAiSuggestionRuleImportJson();
            int count = aiMaterialSuggestionRuleService.importFromConfig(json, AiMaterialSuggestionConstants.SYSTEM_OPERATOR, "JOB");
            log.info("AiMaterialSuggestionRuleImportJob import completed, processed {} rules", count);
            return ReturnT.SUCCESS;
        } catch (Exception e) {
            log.error("AiMaterialSuggestionRuleImportJob failed", e);
            return new ReturnT<>(ReturnT.FAIL_CODE, "Job failed: " + e.getMessage());
        } finally {
            MDCUtil.clear();
        }
    }
}
