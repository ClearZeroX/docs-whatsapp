```java
package com.opay.occ.mcp.businessdiagnostic.core;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/**
 * Immutable log diagnostic environment: MySQL connection plus Apollo coordinates.
 *
 * <p>Secrets are never stored in the profile file; they are resolved from
 * environment variables at call time.</p>
 */
public final class LogEnv {

    private final String name;
    private final String mysqlHost;
    private final int mysqlPort;
    private final String mysqlDatabase;
    private final String mysqlUser;
    private final String mysqlPasswordEnv;
    private final String apolloAppId;
    private final String apolloMeta;
    private final List<String> apolloNamespaces;
    private final String apolloAccessKeyEnv;

    private LogEnv(Builder builder) {
        this.name = builder.name;
        this.mysqlHost = builder.mysqlHost;
        this.mysqlPort = builder.mysqlPort;
        this.mysqlDatabase = builder.mysqlDatabase;
        this.mysqlUser = builder.mysqlUser;
        this.mysqlPasswordEnv = builder.mysqlPasswordEnv;
        this.apolloAppId = builder.apolloAppId;
        this.apolloMeta = builder.apolloMeta;
        this.apolloNamespaces = Collections.unmodifiableList(new ArrayList<>(builder.apolloNamespaces));
        this.apolloAccessKeyEnv = builder.apolloAccessKeyEnv;
    }

    public String getName() {
        return name;
    }

    public String getMysqlHost() {
        return mysqlHost;
    }

    public int getMysqlPort() {
        return mysqlPort;
    }

    public String getMysqlDatabase() {
        return mysqlDatabase;
    }

    public String getMysqlUser() {
        return mysqlUser;
    }

    public String getMysqlPasswordEnv() {
        return mysqlPasswordEnv;
    }

    /** Resolves the MySQL password from its environment variable, or null when unset. */
    public String getMysqlPassword() {
        if (mysqlPasswordEnv == null || mysqlPasswordEnv.isEmpty()) {
            return null;
        }
        return System.getenv(mysqlPasswordEnv);
    }

    public String getApolloAppId() {
        return apolloAppId;
    }

    public String getApolloMeta() {
        return apolloMeta;
    }

    public List<String> getApolloNamespaces() {
        return apolloNamespaces;
    }

    public String getApolloAccessKeyEnv() {
        return apolloAccessKeyEnv;
    }

    /** Resolves the Apollo access key secret, or null when not configured/unset. */
    public String getApolloAccessKey() {
        if (apolloAccessKeyEnv == null || apolloAccessKeyEnv.isEmpty()) {
            return null;
        }
        String value = System.getenv(apolloAccessKeyEnv);
        return (value == null || value.isEmpty()) ? null : value;
    }

    public static Builder builder(String name) {
        return new Builder(name);
    }

    public static final class Builder {
        private final String name;
        private String mysqlHost;
        private int mysqlPort = 3306;
        private String mysqlDatabase;
        private String mysqlUser;
        private String mysqlPasswordEnv;
        private String apolloAppId;
        private String apolloMeta;
        private List<String> apolloNamespaces = Collections.emptyList();
        private String apolloAccessKeyEnv;

        private Builder(String name) {
            this.name = name;
        }

        public Builder mysql(String host, int port, String database, String user, String passwordEnv) {
            this.mysqlHost = host;
            this.mysqlPort = port;
            this.mysqlDatabase = database;
            this.mysqlUser = user;
            this.mysqlPasswordEnv = passwordEnv;
            return this;
        }

        public Builder apollo(String appId, String meta, List<String> namespaces, String accessKeyEnv) {
            this.apolloAppId = appId;
            this.apolloMeta = meta;
            this.apolloNamespaces = namespaces == null ? Collections.<String>emptyList() : namespaces;
            this.apolloAccessKeyEnv = accessKeyEnv;
            return this;
        }

        public LogEnv build() {
            return new LogEnv(this);
        }
    }
}
```

```java
package com.opay.occ.mcp.businessdiagnostic.core;

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;

/**
 * Parsed {@code logDiagnostic} configuration section.
 *
 * <p>Keeps the default environment, the log table name, the service alias map
 * (for example {@code api -> whatsapp-crm-api}) and the environment definitions.</p>
 */
public final class LogConfig {

    public static final String DEFAULT_LOG_TABLE = "test_log";

    private final String defaultEnvironment;
    private final String logTable;
    private final Map<String, String> services;
    private final Map<String, LogEnv> environments;

    private LogConfig(String defaultEnvironment, String logTable,
                      Map<String, String> services, Map<String, LogEnv> environments) {
        this.defaultEnvironment = defaultEnvironment;
        this.logTable = logTable;
        this.services = Collections.unmodifiableMap(new LinkedHashMap<>(services));
        this.environments = Collections.unmodifiableMap(new LinkedHashMap<>(environments));
    }

    public static LogConfig empty() {
        return new LogConfig(null, DEFAULT_LOG_TABLE,
                Collections.<String, String>emptyMap(), Collections.<String, LogEnv>emptyMap());
    }

    public static LogConfig of(String defaultEnvironment, String logTable,
                               Map<String, String> services, Map<String, LogEnv> environments) {
        return new LogConfig(
                defaultEnvironment,
                (logTable == null || logTable.trim().isEmpty()) ? DEFAULT_LOG_TABLE : logTable.trim(),
                services == null ? Collections.<String, String>emptyMap() : services,
                environments == null ? Collections.<String, LogEnv>emptyMap() : environments);
    }

    public String getDefaultEnvironment() {
        return defaultEnvironment;
    }

    public String getLogTable() {
        return logTable;
    }

    public Map<String, String> getServices() {
        return services;
    }

    public Map<String, LogEnv> getEnvironments() {
        return environments;
    }

    /** Distinct full service names in declaration order. */
    public List<String> allServiceNames() {
        return new ArrayList<>(new LinkedHashSet<>(services.values()));
    }

    /**
     * Resolves a user supplied service value to concrete {@code service_name} values.
     * Blank input means "all configured services".
     */
    public List<String> resolveServices(String serviceOrAlias) {
        if (serviceOrAlias == null || serviceOrAlias.trim().isEmpty()) {
            List<String> all = allServiceNames();
            if (all.isEmpty()) {
                throw new IllegalArgumentException("No services configured in logDiagnostic.services");
            }
            return all;
        }
        String value = serviceOrAlias.trim();
        String alias = services.get(value.toLowerCase());
        if (alias != null) {
            return Collections.singletonList(alias);
        }
        if (services.containsValue(value)) {
            return Collections.singletonList(value);
        }
        throw new IllegalArgumentException("Unknown service: " + value + ". Available aliases: "
                + services.keySet() + " (full names: " + allServiceNames() + ")");
    }

    /** Resolves an environment by name, falling back to the default or the single configured one. */
    public LogEnv requireEnvironment(String name) {
        if (environments.isEmpty()) {
            throw new IllegalArgumentException(
                    "No log diagnostic environment configured; add a logDiagnostic section to profiles.json");
        }
        String target = name;
        if (target == null || target.trim().isEmpty()) {
            target = defaultEnvironment;
        }
        if (target == null || target.trim().isEmpty()) {
            if (environments.size() == 1) {
                return environments.values().iterator().next();
            }
            throw new IllegalArgumentException("Environment is required; available: " + environments.keySet());
        }
        LogEnv env = environments.get(target.trim());
        if (env == null) {
            throw new IllegalArgumentException("Unknown environment: " + target + "; available: " + environments.keySet());
        }
        return env;
    }
}
```

```java
package com.opay.occ.mcp.businessdiagnostic.tools;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Extracts trace ids from log content lines.
 *
 * <p>The log pattern embeds the MDC trace id as {@code traceId:[...]}, which is the
 * only reliable link between a business identifier and the full request log trail.</p>
 */
public final class TraceIdExtractor {

    private static final Pattern TRACE_ID_PATTERN = Pattern.compile("traceId:\\[([^\\]]*)\\]");

    private TraceIdExtractor() {
    }

    public static String extract(String content) {
        if (content == null) {
            return null;
        }
        Matcher matcher = TRACE_ID_PATTERN.matcher(content);
        if (!matcher.find()) {
            return null;
        }
        String value = matcher.group(1).trim();
        return value.isEmpty() ? null : value;
    }

    /** Returns the most frequent trace ids, capped at {@code max}. */
    public static List<String> topTraceIds(List<String> contents, int max) {
        List<String> result = new ArrayList<>();
        if (contents == null || contents.isEmpty() || max <= 0) {
            return result;
        }
        Map<String, Integer> counts = new LinkedHashMap<>();
        for (String content : contents) {
            String traceId = extract(content);
            if (traceId == null) {
                continue;
            }
            counts.put(traceId, counts.containsKey(traceId) ? counts.get(traceId) + 1 : 1);
        }
        List<Map.Entry<String, Integer>> entries = new ArrayList<>(counts.entrySet());
        entries.sort(new Comparator<Map.Entry<String, Integer>>() {
            @Override
            public int compare(Map.Entry<String, Integer> left, Map.Entry<String, Integer> right) {
                int byCount = right.getValue().compareTo(left.getValue());
                return byCount != 0 ? byCount : left.getKey().compareTo(right.getKey());
            }
        });
        for (Map.Entry<String, Integer> entry : entries) {
            if (result.size() >= max) {
                break;
            }
            result.add(entry.getKey());
        }
        return result;
    }
}
```

```java
    public static Connection openReadOnly(String host, int port, String database,
                                          String user, String password) throws SQLException {
        String url = "jdbc:mysql://" + host + ":" + port + "/" + database
                + "?useUnicode=true&characterEncoding=utf8&useSSL=false&serverTimezone=Asia/Shanghai"
                + "&connectTimeout=3000&socketTimeout=5000";
        Connection connection = DriverManager.getConnection(url, user, password);
        try (Statement statement = connection.createStatement()) {
            statement.execute("SET SESSION TRANSACTION READ ONLY");
        }
        connection.setReadOnly(true);
        return connection;
    }
```

```java
package com.opay.occ.mcp.businessdiagnostic.tools;

import com.opay.occ.mcp.businessdiagnostic.core.LogConfig;
import com.opay.occ.mcp.businessdiagnostic.core.LogEnv;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.text.SimpleDateFormat;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.Date;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.TimeUnit;

/**
 * Log diagnostics against the {@code test_log} table.
 *
 * <p>Only structured conditions are accepted; the SQL is generated by
 * {@link SqlBuilder} and every value is bound as a JDBC parameter.</p>
 */
public final class LogTools {

    public static final int DEFAULT_WINDOW_MINUTES = 20;
    public static final int DEFAULT_LIMIT = 200;
    public static final int MAX_LIMIT = 1000;
    public static final int MAX_TRACE_IDS = 20;

    private static final int QUERY_TIMEOUT_SECONDS = 5;
    private static final String LOG_TIME_FORMAT = "yyyy-MM-dd HH:mm:ss.SSS";
    private static final List<String> LOG_COLUMNS = Arrays.asList(
            "id", "service_name", "pod", "log_time", "trace_id", "log_level", "content");

    private LogTools() {
    }

    /** Tool: list_log_environments */
    public static Map<String, Object> listEnvironments(LogConfig config) {
        List<Map<String, Object>> environments = new ArrayList<>();
        for (LogEnv env : config.getEnvironments().values()) {
            Map<String, Object> item = new LinkedHashMap<>();
            item.put("name", env.getName());
            item.put("mysqlHost", env.getMysqlHost());
            item.put("mysqlDatabase", env.getMysqlDatabase());
            item.put("apolloAppId", env.getApolloAppId());
            item.put("apolloMeta", env.getApolloMeta());
            item.put("apolloNamespaces", env.getApolloNamespaces());
            environments.add(item);
        }
        Map<String, Object> response = new LinkedHashMap<>();
        response.put("ok", true);
        response.put("defaultEnvironment", config.getDefaultEnvironment());
        response.put("logTable", config.getLogTable());
        response.put("services", new LinkedHashMap<>(config.getServices()));
        response.put("environments", environments);
        return response;
    }

    /** Tool: query_logs */
    public static Map<String, Object> query(LogConfig config, String env, String service,
                                            String traceId, String keyword, String logLevel,
                                            Long timeStart, Long timeEnd, int limit, String order) {
        if (isBlank(traceId) && isBlank(keyword)) {
            throw new IllegalArgumentException("Either traceId or keyword is required");
        }
        LogEnv logEnv = config.requireEnvironment(env);
        List<String> services = config.resolveServices(service);
        String table = config.getLogTable();
        long[] window = resolveTimeWindow(timeStart, timeEnd, System.currentTimeMillis());
        List<SqlBuilder.SqlQuery> queries = buildQuery(
                table, services, traceId, keyword, logLevel, window[0], window[1], limit, order);

        String password = logEnv.getMysqlPassword();
        if (password == null || password.isEmpty()) {
            throw new IllegalStateException(
                    "Missing password environment variable: " + logEnv.getMysqlPasswordEnv());
        }

        List<Map<String, Object>> rows = new ArrayList<>();
        try (Connection connection = JdbcSupport.openReadOnly(
                logEnv.getMysqlHost(), logEnv.getMysqlPort(), logEnv.getMysqlDatabase(),
                logEnv.getMysqlUser(), password)) {
            for (SqlBuilder.SqlQuery query : queries) {
                rows.addAll(executeQuery(connection, query));
            }
        } catch (SQLException e) {
            throw new IllegalStateException("MySQL log query failed: " + e.getMessage(), e);
        }

        boolean hasMore = rows.size() > limit;
        List<Map<String, Object>> page = hasMore ? new ArrayList<>(rows.subList(0, limit)) : rows;
        List<String> contents = new ArrayList<>();
        Map<String, Integer> serviceHits = new LinkedHashMap<>();
        for (Map<String, Object> row : page) {
            Object content = row.get("content");
            contents.add(content == null ? null : content.toString());
            String serviceName = String.valueOf(row.get("serviceName"));
            serviceHits.put(serviceName, serviceHits.containsKey(serviceName) ? serviceHits.get(serviceName) + 1 : 1);
        }

        Map<String, Object> response = new LinkedHashMap<>();
        response.put("ok", true);
        response.put("env", logEnv.getName());
        response.put("table", table);
        response.put("services", services);
        response.put("traceId", traceId);
        response.put("keyword", keyword);
        response.put("timeStart", format(window[0]));
        response.put("timeEnd", format(window[1]));
        response.put("rows", page);
        response.put("rowCount", page.size());
        response.put("serviceHits", serviceHits);
        response.put("matchedTraceIds", TraceIdExtractor.topTraceIds(contents, MAX_TRACE_IDS));
        response.put("hasMore", hasMore);
        response.put("queryCount", queries.size());
        return response;
    }

    /** Defaults to the last {@value #DEFAULT_WINDOW_MINUTES} minutes when no time is provided. */
    static long[] resolveTimeWindow(Long timeStart, Long timeEnd, long nowMillis) {
        long end = timeEnd != null ? timeEnd : nowMillis;
        long start = timeStart != null ? timeStart : end - TimeUnit.MINUTES.toMillis(DEFAULT_WINDOW_MINUTES);
        if (start > end) {
            throw new IllegalArgumentException("timeStart must be less than or equal to timeEnd");
        }
        return new long[]{start, end};
    }

    static List<SqlBuilder.SqlQuery> buildQuery(String table, List<String> services, String traceId,
                                                String keyword, String logLevel, long startMillis,
                                                long endMillis, int limit, String order) {
        if (limit < 1 || limit > MAX_LIMIT) {
            throw new IllegalArgumentException("limit must be between 1 and " + MAX_LIMIT);
        }
        if (services == null || services.isEmpty()) {
            throw new IllegalArgumentException("services must not be empty");
        }

        List<Object> serviceValues = new ArrayList<>(services);
        List<Map<String, Object>> where = new ArrayList<>();
        where.add(condition("service_name", "in", serviceValues));
        if (!isBlank(traceId)) {
            where.add(condition("trace_id", "eq", traceId.trim()));
        }
        if (!isBlank(logLevel)) {
            where.add(condition("log_level", "eq", logLevel.trim().toUpperCase()));
        }
        if (!isBlank(keyword)) {
            where.add(condition("content", "like", "%" + keyword.trim() + "%"));
        }
        // MySQL compares DATETIME columns correctly against formatted string literals;
        // passing epoch millis would be converted into YYYYMMDDHHMMSS and break the range.
        where.add(condition("log_time", "gte", format(startMillis)));
        where.add(condition("log_time", "lte", format(endMillis)));

        List<Map<String, Object>> orderBy = Collections.singletonList(
                orderBy("log_time", "desc".equalsIgnoreCase(order) ? "desc" : "asc"));

        return SqlBuilder.buildSelect(table, LOG_COLUMNS, where, orderBy, limit, 0);
    }

    private static List<Map<String, Object>> executeQuery(Connection connection, SqlBuilder.SqlQuery query)
            throws SQLException {
        List<Map<String, Object>> rows = new ArrayList<>();
        try (PreparedStatement statement = connection.prepareStatement(query.getSql())) {
            statement.setQueryTimeout(QUERY_TIMEOUT_SECONDS);
            List<Object> params = query.getParams();
            for (int i = 0; i < params.size(); i++) {
                statement.setObject(i + 1, params.get(i));
            }
            try (ResultSet resultSet = statement.executeQuery()) {
                while (resultSet.next()) {
                    Map<String, Object> row = new LinkedHashMap<>();
                    row.put("id", resultSet.getLong("id"));
                    row.put("serviceName", resultSet.getString("service_name"));
                    row.put("pod", resultSet.getString("pod"));
                    long logTimeMillis = resultSet.getTimestamp("log_time") == null
                            ? 0L : resultSet.getTimestamp("log_time").getTime();
                    row.put("logTime", format(logTimeMillis));
                    row.put("logTimeMillis", logTimeMillis);
                    row.put("traceId", resultSet.getString("trace_id"));
                    row.put("logLevel", resultSet.getString("log_level"));
                    row.put("content", resultSet.getString("content"));
                    rows.add(row);
                }
            }
        }
        return rows;
    }

    private static Map<String, Object> condition(String field, String op, Object value) {
        Map<String, Object> condition = new LinkedHashMap<>();
        condition.put("field", field);
        condition.put("op", op);
        condition.put("value", value);
        return condition;
    }

    private static Map<String, Object> orderBy(String field, String direction) {
        Map<String, Object> order = new LinkedHashMap<>();
        order.put("field", field);
        order.put("direction", direction);
        return order;
    }

    private static String format(long millis) {
        return new SimpleDateFormat(LOG_TIME_FORMAT).format(new Date(millis));
    }

    private static boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }
}
```

`src/main/java/com/opay/occ/mcp/businessdiagnostic/tools/ApolloTools.java`

```java
package com.opay.occ.mcp.businessdiagnostic.tools;

import com.ctrip.framework.apollo.Config;
import com.ctrip.framework.apollo.ConfigService;
import com.opay.occ.mcp.businessdiagnostic.core.LogConfig;
import com.opay.occ.mcp.businessdiagnostic.core.LogEnv;

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeSet;

/**
 * Reads Apollo configuration with the apollo-client SDK, mirroring
 * {@code TestLogServiceImpl#getAllApolloConfigs} so the values match what the
 * running service actually sees.
 *
 * <p>Apollo Client keeps a process-wide client configuration, so credentials are
 * applied once before the first {@code ConfigService} call.</p>
 *
 * <p>TODO(multi-environment): support more than one environment at a time. The meta
 * server and app id are global to the SDK process, so multiple environments would
 * need dedicated Apollo client instances created from a custom client config.</p>
 */
public final class ApolloTools {

    private static final String COMMENT_UNAVAILABLE = "";

    private ApolloTools() {
    }

    /** Tool: get_apollo_configs */
    public static Map<String, Object> getConfigs(LogConfig config, String env, String namespace, String keyKeyword) {
        LogEnv logEnv = config.requireEnvironment(env);
        if (logEnv.getApolloAppId() == null || logEnv.getApolloMeta() == null) {
            throw new IllegalStateException(
                    "Apollo is not configured for environment: " + logEnv.getName());
        }
        applyCredentials(logEnv);

        List<String> namespaces = resolveNamespaces(logEnv, namespace);
        List<Map<String, Object>> items = new ArrayList<>();
        List<String> warnings = new ArrayList<>();

        for (String current : namespaces) {
            try {
                Config apolloConfig = ConfigService.getConfig(current);
                if (apolloConfig == null) {
                    warnings.add("namespace not available: " + current);
                    continue;
                }
                for (String key : new TreeSet<>(apolloConfig.getPropertyNames())) {
                    if (!matchesKey(key, keyKeyword)) {
                        continue;
                    }
                    Map<String, Object> item = new LinkedHashMap<>();
                    item.put("namespace", current);
                    item.put("key", key);
                    item.put("value", apolloConfig.getProperty(key, ""));
                    // The runtime client does not expose configuration comments.
                    item.put("comment", COMMENT_UNAVAILABLE);
                    items.add(item);
                }
            } catch (Exception e) {
                warnings.add("namespace failed: " + current + " (" + e.getMessage() + ")");
            }
        }

        Map<String, Object> response = new LinkedHashMap<>();
        response.put("ok", true);
        response.put("env", logEnv.getName());
        response.put("appId", logEnv.getApolloAppId());
        response.put("meta", logEnv.getApolloMeta());
        response.put("namespaces", namespaces);
        response.put("items", items);
        response.put("itemCount", items.size());
        response.put("warnings", warnings);
        return response;
    }

    /** Applies apollo-client credentials as system properties. */
    static void applyCredentials(LogEnv env) {
        System.setProperty("app.id", env.getApolloAppId());
        System.setProperty("apollo.meta", env.getApolloMeta());
        String accessKey = env.getApolloAccessKey();
        if (accessKey != null) {
            System.setProperty("apollo.accesskey.secret", accessKey);
        }
    }

    static List<String> resolveNamespaces(LogEnv env, String namespace) {
        if (namespace == null || namespace.trim().isEmpty()) {
            return new ArrayList<>(env.getApolloNamespaces());
        }
        return Collections.singletonList(namespace.trim());
    }

    static boolean matchesKey(String key, String keyKeyword) {
        if (keyKeyword == null || keyKeyword.trim().isEmpty()) {
            return true;
        }
        return key != null && key.toLowerCase().contains(keyKeyword.trim().toLowerCase());
    }
}
```

```markdown
# Log Diagnostics

Investigate application logs stored in the `test_log` table for the whatsapp-crm services.

## Before you start

- Logs are retained for about 2 hours; older records do not exist.
- The four services are `whatsapp-crm-api` (api), `whatsapp-crm-job` (job), `whatsapp-crm-mq` (mq), `whatsapp-crm-otp` (otp).
- Business tables do not store trace ids, so a business identifier (phone number, task id, template name) is found by searching log content.

## Workflow

1. Call `list_log_environments` when you need the environment name or service aliases.
2. Call `query_logs` with:
   - `traceId` when the user provides one;
   - `keyword` when the user gives a phone number, task id, template name, or error text;
   - both when you already know the trace id you want to narrow down.
3. Omit `service` when you cannot tell which service is involved: all four services are then queried.
4. Omit the time range only when the default (last 20 minutes) is acceptable; otherwise pass `timeStart`/`timeEnd` in epoch milliseconds.
5. If the result contains `matchedTraceIds`, pick the relevant one and call `query_logs` again with that `traceId` to read the full request trail.
6. Summarize the timeline for the user; quote only the log lines that matter.

Call `get_apollo_configs` when the investigation depends on a feature flag or a configuration value.
Filter with `keyKeyword` to avoid dumping every property. Values come from the same
apollo-client path the application uses, so they reflect the running service.

## Safety

- The tools are read-only; never attempt to write to the database.
- Do not ask the user for passwords in chat; credentials come from environment variables.
- Log content is returned unmasked; avoid pasting unrelated personal data into the summary.
- Keep result sets small: use `limit` and a narrow time range.
```
